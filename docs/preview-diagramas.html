4. Modelo Entidade-Relacionamento

Um motorista conduz uma rota; a rota tem vários pontos de embarque e vários alunos; cada aluno pertence a um ponto; cada registro de embarque referencia o aluno e o ponto correspondente.

erDiagram
    MOTORISTA ||--o| ROTA : "conduz"
    ROTA ||--o{ PONTO_EMBARQUE : "possui"
    ROTA ||--o{ ALUNO : "atende"
    PONTO_EMBARQUE ||--o{ ALUNO : "recebe"
    ALUNO ||--o{ REGISTRO_EMBARQUE : "possui"
    PONTO_EMBARQUE ||--o{ REGISTRO_EMBARQUE : "referencia"

    MOTORISTA {
        int id_motorista PK
        string nome
        string cnh
        string telefone
    }

    ROTA {
        int id_rota PK
        string nome_rota
        string turno
        int id_motorista FK
    }

    PONTO_EMBARQUE {
        int id_ponto PK
        string nome_ponto
        string endereco
        int id_rota FK
    }

    ALUNO {
        int id_aluno PK
        string nome
        string turma
        int id_rota FK
        int id_ponto FK
    }

    REGISTRO_EMBARQUE {
        int id_registro PK
        int id_aluno FK
        int id_ponto FK
        date data
        boolean embarcou
    }
